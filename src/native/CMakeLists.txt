cmake_minimum_required(VERSION 3.20)
project(dv_tape_capture VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
if(WIN32 AND NOT MINGW)
    message(FATAL_ERROR "Windows backend requires MinGW-w64 and winpthreads")
endif()

set(LIBUSB_INCLUDE_DIR "" CACHE PATH "Directory containing libusb.h")
set(LIBUSB_LIBRARY "" CACHE FILEPATH "libusb 1.0.30 static or import library")
if(LIBUSB_INCLUDE_DIR AND LIBUSB_LIBRARY)
    add_library(dv_libusb UNKNOWN IMPORTED)
    set_target_properties(dv_libusb PROPERTIES IMPORTED_LOCATION "${LIBUSB_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${LIBUSB_INCLUDE_DIR}")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(USB REQUIRED IMPORTED_TARGET libusb-1.0>=1.0.30)
    add_library(dv_libusb ALIAS PkgConfig::USB)
endif()
find_package(Threads REQUIRED)

function(configure_capture target)
    target_link_libraries(${target} PRIVATE dv_libusb Threads::Threads)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(${target} PRIVATE _FILE_OFFSET_BITS=64)
    if(WIN32)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WINVER=0x0603 _WIN32_WINNT=0x0603)
        target_link_libraries(${target} PRIVATE "-Wl,-Bstatic" winpthread "-Wl,-Bdynamic")
    elseif(APPLE)
        target_link_libraries(${target} PRIVATE "-framework IOKit" "-framework CoreFoundation"
            "-framework Security" objc)
    endif()
endfunction()

add_executable(uvcdv-capture uvcdv-capture.c capture_platform.c)
configure_capture(uvcdv-capture)
if(WIN32)
    target_link_options(uvcdv-capture PRIVATE -municode)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test-capture test_capture.c capture_platform.c)
    configure_capture(test-capture)
    add_test(NAME callback-and-control COMMAND test-capture)
    add_executable(test-platform test_platform.c capture_platform.c)
    target_link_libraries(test-platform PRIVATE Threads::Threads)
    target_compile_definitions(test-platform PRIVATE _FILE_OFFSET_BITS=64)
    if(WIN32)
        target_compile_definitions(test-platform PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0603)
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/synthetic-tests")
    add_test(NAME platform-path-and-pipe COMMAND test-platform "${CMAKE_CURRENT_BINARY_DIR}/synthetic-tests")
endif()

install(TARGETS uvcdv-capture RUNTIME DESTINATION bin/native)
